public static class RaidMapGenerator {
 public static Dictionary<Vector2Int,int> GetDistances(StageMapLayout map,Vector2Int start){
  var reached=new Dictionary<Vector2Int,int>{{start,0}};var queue=new Queue<Vector2Int>();queue.Enqueue(start);
  while(queue.Count>0){var c=queue.Dequeue();var room=map.rooms.Find(x=>x.coord==c);if(room==null)continue;
   for(int i=0;i<4;i++){var next=c+LandmarkMapGenerator.Steps[i];var other=map.rooms.Find(x=>x.coord==next);
    if(other!=null&&room.HasConnection(LandmarkMapGenerator.Flags[i])&&other.HasConnection(LandmarkMapGenerator.Flags[i^1])&&!reached.ContainsKey(next)){reached.Add(next,reached[c]+1);queue.Enqueue(next);}}}
  return reached;
 }
}
public static class SystemsPatchChecks {
 private static int count;
 private static void Check(bool yes,string name){if(!yes)throw new Exception("FAIL: "+name);count++;}
 private static void Near(float value,float expected,string name)=>Check(Math.Abs(value-expected)<.0001f,name);
 public static int Run(){
  count=0;
  Near(RaidArmorPenetration.Resolve(100,0,100,100,1).probability,.05f,"minimum probability");
  Near(RaidArmorPenetration.Resolve(100,100,100,100,1).probability,.95f,"equal probability");
  Near(RaidArmorPenetration.Resolve(100,200,1,100,1).probability,.95f,"overpenetration cap");
  Near(RaidArmorPenetration.Resolve(0,0,1,1,1).healthMultiplier,1,"unarmored");
  Near(RaidArmorPenetration.Resolve(100,50,1,100,0).healthMultiplier,.75f,"successful damage");
  Near(RaidArmorPenetration.Resolve(100,50,1,100,1).healthMultiplier,.15f,"failed damage");
  float previous=0;
  for(int i=0;i<=100;i++){float c=RaidArmorPenetration.Resolve(100,i,1,1,0).probability;Check(c>=previous,"monotonic "+i);previous=c;}
  Near(RaidArmorPenetration.Resolve(100,50,1,100,0).probability,.2504301f,"k2.5 midpoint");
  Near(CombatBalanceRules.AmmoDamageAdjustment(BulletTier.Tier1),0,"T1 damage");
  Near(CombatBalanceRules.AmmoDamageAdjustment(BulletTier.Tier6),-6,"T6 adjustment");
  Check(CombatBalanceRules.Penetration(8)==85,"T8 penetration");
  Near(CombatBalanceRules.EnemyHealth(3,EnemyShooterRole.Boss,1),650,"S3 boss");
  Near(CombatBalanceRules.EnemyHealth(4,EnemyShooterRole.StandardEnemy,1),300,"S4 normal");
  Near(CombatBalanceRules.EnemyDamage(4,EnemyShooterRole.Boss,1),52,"S4 boss damage");
  Check(FactionDiscoveryRulesSO.InRange(new(0,0,0),new(45,100,0),45),"planar sharing boundary");
  Check(!FactionDiscoveryRulesSO.InRange(new(0,0,0),new(46,0,0),45),"outside sharing");
  Check(!FactionDiscoveryRulesSO.InRange(new(0,0,0),new(0,0,0),0),"disabled radius");
  var catalog=new CraftingCatalogSO();
  foreach(var rarity in new[]{PlayerShopItemRarity.Normal,PlayerShopItemRarity.Rare,PlayerShopItemRarity.Unique,PlayerShopItemRarity.Legendary,PlayerShopItemRarity.Mythic})
   catalog.materials.Add(new ValuableItemSO{Guid="material-"+rarity,Rarity=rarity,ValuableCategoryId="military"});
  var recipe=new CraftingRecipe{id="test",output=new PassiveItemSO(),randomModuleMaterials=true,valuableCategories=new[]{"military"}};
  foreach(var rarity in new[]{PlayerShopItemRarity.Normal,PlayerShopItemRarity.Rare,PlayerShopItemRarity.Unique,PlayerShopItemRarity.Legendary,PlayerShopItemRarity.Mythic}){
   recipe.moduleRarity=rarity;
   Check(CraftingRecipeRoll.TryRoll(catalog,recipe,5,out var state,out var message),"roll "+rarity+": "+message);
   Check(state.materials.Sum(x=>x.amount)==CraftingRecipeRoll.RequiredRarities(rarity).Length,"recipe total "+rarity);
   Check(state.revision==5,"revision");
   var expected=CraftingRecipeRoll.RequiredRarities(rarity).GroupBy(x=>x).ToDictionary(x=>"material-"+x.Key,x=>x.Count());
   Check(state.materials.All(x=>expected[x.itemGuid]==x.amount),"no rarity downgrade "+rarity);
  }
  var approved=catalog.materials[0];catalog.materials.RemoveAt(0);recipe.moduleRarity=PlayerShopItemRarity.Normal;
  Check(!CraftingRecipeRoll.TryRoll(catalog,recipe,6,out var rejected,out _)&&rejected==null,"shortage no recipe");
  catalog.materials.Add(approved);
  var profile=new Save();profile.craftingRecipes.Add(new CraftRecipeSave{recipeId="test",revision=1});
  var run=new RunSaveData();run.craftingRecipeCommits.Add(new CraftRecipeSave{recipeId="test",revision=2,
   materials=new(){new(){itemGuid="material-Normal",amount=3}}});
  CraftingRecipeRoll.ReconcileCommittedRecipes(profile,run);Check(profile.craftingRecipes[0].revision==2,"recover interrupted profile");
  CraftingRecipeRoll.ReconcileCommittedRecipes(profile,run);Check(profile.craftingRecipes.Count==1,"recovery idempotence");
  run.craftingRecipeCommits[0].revision=0;CraftingRecipeRoll.ReconcileCommittedRecipes(profile,run);
  Check(profile.craftingRecipes[0].revision==2,"never rollback committed recipe");
  Check(!CraftingRecipeRoll.IsValidState(new CraftRecipeSave{recipeId="broken"}),"empty saved recipe rejected");
  Check(!CraftingRecipeRoll.IsValidState(new CraftRecipeSave{recipeId="broken",materials=new(){null}}),"null saved material rejected");
  Check(!CraftingRecipeRoll.IsValidState(new CraftRecipeSave{recipeId="broken",materials=new(){
   new(){itemGuid="a",amount=1},new(){itemGuid="a",amount=2}}}),"duplicate saved material rejected");
  var fixedRecipe=new CraftingRecipe{id="overflow",output=new ItemSO(),fixedMaterials=new[]{
   new ShopTradeRequirement(approved,int.MaxValue),new ShopTradeRequirement(approved,1)}};
  Check(!CraftingRecipeRoll.TryRoll(catalog,fixedRecipe,0,out _,out _),"material quantity overflow rejected");
  var stage=new StageDefinitionSO();
  for(int i=0;i<3;i++){
   var definition=new LandmarkDefinitionSO{id="landmark"+i,displayName="L"+i,powerStation=i==0};
   var variant=new LandmarkPrefab();variant.cells.Add(new(){offset=new(0,0)});
   if(i!=1)variant.cells.Add(new(){offset=new(1,0)});
   if(i==2)variant.cells.Add(new(){offset=new(0,1)});
   definition.variants.Add(variant);stage.Landmarks.guaranteed.Add(definition);
  }
  for(int seed=1;seed<=100;seed++){
   bool ok=LandmarkMapGenerator.TryGenerate(stage,seed,out var map,out var message);
   Check(ok,"map "+seed+": "+message);
   Check(map.rooms.Where(x=>!string.IsNullOrEmpty(x.landmarkId)).Select(x=>x.coord).Distinct().Count()==6,"guaranteed footprint "+seed);
   Check(map.rooms.Where(x=>!string.IsNullOrEmpty(x.landmarkId)).All(x=>x.isRevealed),"public landmarks "+seed);
   Check(!string.IsNullOrEmpty(map.rooms.Find(x=>x.coord==map.bossRoomCoord).landmarkId),"boss in landmark "+seed);
   Check(map.extractionRooms.All(c=>{var room=map.rooms.Find(x=>x.coord==c);return LandmarkMapGenerator.Flags.Count(f=>room.HasConnection(f))==1;}),"leaf extraction "+seed);
   Check(LandmarkMapGenerator.TryValidate(map,out _),"saved validation "+seed);
   Check(LandmarkMapGenerator.TryGenerate(stage,seed,out var duplicate,out _),"repeat seed");
   Check(string.Join(";",map.rooms.Select(x=>x.coord+":"+x.landmarkId))==string.Join(";",duplicate.rooms.Select(x=>x.coord+":"+x.landmarkId)),"deterministic "+seed);
  }
  foreach(int size in new[]{7,9})foreach(int exitCount in new[]{1,2,3})
  {
   stage.RoomGridSize=size;stage.RaidMap.extractionCount=exitCount;
   for(int seed=1;seed<=100;seed++)
   {
    Check(LandmarkMapGenerator.TryGenerate(stage,seed,out var map,out var error),"grid/exit/seed "+size+"/"+exitCount+"/"+seed+": "+error);
    Check(map.extractionPortals.Count==exitCount,"exact extraction route count");
   }
  }
  stage.RoomGridSize=1;
  Check(!LandmarkMapGenerator.TryGenerate(stage,1,out _,out _),"small grid explicit failure");
  Console.WriteLine("Systems patch managed checks passed: "+count);
  return count;
 }
